cmake_minimum_required(VERSION 3.16)
project(iss_water LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(gz-sim8 REQUIRED)
find_package(gz-rendering8 REQUIRED COMPONENTS ogre2)
find_package(gz-gui8 REQUIRED)
find_package(gz-transport13 REQUIRED)
find_package(gz-plugin2 REQUIRED COMPONENTS register)
find_package(gz-math7 REQUIRED)
find_package(Qt5 REQUIRED COMPONENTS Core Gui Qml Quick)
cmake_policy(SET CMP0100 NEW)
find_path(ISS_OGRE_INCLUDE OgreRoot.h PATH_SUFFIXES include/OGRE-Next include/OGRE-2.3 REQUIRED)
find_library(ISS_OGRE_MAIN OgreNextMain PATH_SUFFIXES lib lib/OGRE-Next lib/OGRE-2.3 REQUIRED)
find_library(ISS_OGRE_PBS OgreNextHlmsPbs PATH_SUFFIXES lib lib/OGRE-Next lib/OGRE-2.3 REQUIRED)
add_library(iss-ogre INTERFACE)
target_include_directories(iss-ogre INTERFACE ${ISS_OGRE_INCLUDE}
  ${ISS_OGRE_INCLUDE}/Hlms/Pbs ${ISS_OGRE_INCLUDE}/Hlms/Unlit ${ISS_OGRE_INCLUDE}/Hlms/Common)
target_link_libraries(iss-ogre INTERFACE ${ISS_OGRE_MAIN} ${ISS_OGRE_PBS})
add_library(water-render OBJECT WaterRender.cc)
target_link_libraries(water-render PUBLIC gz-rendering8::ogre2 iss-ogre)
add_library(iss-water-system SHARED WaterSystem.cc $<TARGET_OBJECTS:water-render>)
target_link_libraries(iss-water-system PRIVATE gz-sim8::gz-sim8 gz-transport13::gz-transport13 gz-plugin2::register gz-rendering8::ogre2 iss-ogre)
add_library(WaterGui SHARED WaterGui.cc WaterGui.hh WaterGui.qrc $<TARGET_OBJECTS:water-render>)
target_link_libraries(WaterGui PRIVATE gz-gui8::gz-gui8 gz-transport13::gz-transport13
  gz-rendering8::ogre2 iss-ogre Qt5::Core Qt5::Gui Qt5::Qml Qt5::Quick)
include(CTest)
if(BUILD_TESTING)
  add_executable(water-adhesion-test TestWaterAdhesion.cc)
  target_link_libraries(water-adhesion-test PRIVATE gz-math7::gz-math7)
  add_test(NAME water-adhesion-profile COMMAND water-adhesion-test)
endif()
